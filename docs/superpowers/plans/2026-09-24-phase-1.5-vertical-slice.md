# Phase 1.5 Vertical Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Real Hacker News signals are fetched, stored, served by an API and visible on a web page.

**Architecture:** A plain `run_ingest` function pulls from a `Source` plugin, stores `RawSignal`s in a `signals` table idempotently (unique `canonical_url`) and records a `stage_runs` row. FastAPI exposes `GET /signals` and a flag-gated `POST /ingest/{source}` over the repository. A Next.js server component reads the API. No queue, no state machine, no LLM.

**Tech Stack:** Python 3.13, SQLAlchemy 2 + Alembic, Pydantic 2, httpx, FastAPI + uvicorn, pytest, Next.js (App Router, TypeScript strict), vitest.

**Spec:** [ADR-0005](../../adr/0005-vertical-slice-before-pipeline-infrastructure.md). PRD §33 (`RawSignal`), §24 (`Source` Protocol), §43 (Phase 1.5).

- Layering: `apps/ → stages/ → core/ ← plugins/`. `core/` imports nothing from `plugins/`, `stages/` or `apps/`. A stage never imports a concrete provider. An SDK or HTTP client for a source is imported only inside that plugin.
- Errors: use the typed hierarchy in `signalforge/core/errors.py`. Every external call has a timeout and bounded retries with backoff. Failures end in a visible `failed` state with a reason.
- Tests: pytest, no external network, no real LLM. Integration tests that need Postgres are marked `integration` and skip unless `TEST_DATABASE_URL` is set. Every bug fix ships a regression test.
- Database: schema changes only through Alembic, backward-compatible. Follow ADR-0004: a branch may run its migration on dev but must `alembic downgrade -1` before switching branches or merging. Prod is migrated only from `main`.
- Fetched web content is untrusted data: stored as data, never interpreted, never used to pick tools or actions; the UI renders it as text and only links `http`/`https` URLs.
- New dependencies are minimal, justified in the PR description and pinned through `uv.lock`.
- Verify commands before claiming done: `uv run ruff check . && uv run ruff format --check . && uv run mypy && uv run pytest`.

```python
"""URL canonicalisation and content hashing for signals.

``canonical_url`` is the dedupe key: two signals about the same page must map to
the same string, otherwise the same event is researched twice (PRD §12).
"""

import hashlib
from urllib.parse import parse_qsl, urlencode, urlsplit, urlunsplit

from signalforge.core.errors import ArtifactValidationError

_DEFAULT_PORTS = {"http": 80, "https": 443}
# Tracking parameters carry no meaning about the page. utm_* is matched by prefix.
_TRACKING_PARAMS = frozenset({"fbclid", "gclid", "ref", "ref_src", "mc_cid", "mc_eid"})


def canonical_url(url: str) -> str:
    """Return a stable form of an http(s) URL.

    Lowercases scheme and host, drops credentials, default ports, fragments,
    tracking parameters and a trailing slash, and sorts the query string.

    Raises:
        ArtifactValidationError: if ``url`` is not an absolute http(s) URL.
    """
    parts = urlsplit(url.strip())
    scheme = parts.scheme.lower()
    try:
        host, port = parts.hostname, parts.port
    except ValueError as exc:
        raise ArtifactValidationError("Invalid URL", context={"url": url}) from exc
    if scheme not in _DEFAULT_PORTS or not host:
        raise ArtifactValidationError("Not an absolute http(s) URL", context={"url": url})

    netloc = host if port in (None, _DEFAULT_PORTS[scheme]) else f"{host}:{port}"
    query = urlencode(
        sorted(
            (key, value)
            for key, value in parse_qsl(parts.query, keep_blank_values=True)
            if not key.lower().startswith("utm_") and key.lower() not in _TRACKING_PARAMS
        )
    )
    return urlunsplit((scheme, netloc, parts.path.rstrip("/"), query, ""))


def content_hash(title: str, summary: str) -> str:
    """Hash of the normalised text, so a re-titled duplicate can be detected later."""
    normalised = "\n".join(" ".join(text.split()).casefold() for text in (title, summary))
    return hashlib.sha256(normalised.encode()).hexdigest()
```

```python
"""RawSignal: what a source plugin emits, before any judgement is applied (PRD §33)."""

from datetime import datetime
from typing import Any, ClassVar, Self

from pydantic import Field, field_validator, model_validator

from signalforge.core.artifacts.base import Artifact
from signalforge.core.canonical import canonical_url, content_hash


class RawSignal(Artifact):
    SCHEMA_VERSION: ClassVar[int] = 1

    source: str = Field(min_length=1, max_length=64)
    # ponytail: free text for now, becomes an enum when a second source type exists.
    source_type: str = Field(min_length=1, max_length=32)
    canonical_url: str
    title: str = Field(min_length=1)
    published_at: datetime
    summary: str = ""
    entities: tuple[str, ...] = ()
    topics: tuple[str, ...] = ()
    metadata: dict[str, Any] = Field(default_factory=dict)
    content_hash: str = ""

    @field_validator("title", "summary", mode="before")
    @classmethod
    def _strip(cls, value: Any) -> Any:
        return value.strip() if isinstance(value, str) else value

    @field_validator("canonical_url")
    @classmethod
    def _canonicalise(cls, value: str) -> str:
        return canonical_url(value)

    @field_validator("published_at")
    @classmethod
    def _require_timezone(cls, value: datetime) -> datetime:
        if value.tzinfo is None:
            raise ValueError("published_at must be timezone-aware")
        return value

    @model_validator(mode="after")
    def _fill_content_hash(self) -> Self:
        if not self.content_hash:
            # Frozen model: object.__setattr__ is the sanctioned way to finish construction.
            object.__setattr__(self, "content_hash", content_hash(self.title, self.summary))
        return self
```

```python
"""Signals: normalised items fetched from sources (PRD §33, §34).

One row per canonical URL. Rows are immutable artifacts, so a refetch of a known
URL is ignored rather than updated.
"""

import uuid
from datetime import datetime
from typing import Any

from sqlalchemy import DateTime, ForeignKey, Index, String, Text, UniqueConstraint
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import Mapped, mapped_column

from signalforge.db.base import Base, SchemaVersionMixin, TimestampsMixin, UUIDPrimaryKeyMixin


class Signal(Base, UUIDPrimaryKeyMixin, TimestampsMixin, SchemaVersionMixin):
    __tablename__ = "signals"
    __table_args__ = (
        UniqueConstraint("canonical_url", name="uq_signals_canonical_url"),
        Index("ix_signals_content_hash", "content_hash"),
        # Serves the keyset pagination order: newest first, id as tiebreaker.
        Index("ix_signals_published_at_id", "published_at", "id"),
    )

    source: Mapped[str] = mapped_column(String(64))
    source_type: Mapped[str] = mapped_column(String(32))
    canonical_url: Mapped[str] = mapped_column(Text)
    title: Mapped[str] = mapped_column(Text)
    published_at: Mapped[datetime] = mapped_column(DateTime(timezone=True))
    summary: Mapped[str] = mapped_column(Text, default="")
    # Promoted to columns only when queried or filtered (PRD §33); until then JSONB.
    entities: Mapped[list[str]] = mapped_column(JSONB, default=list)
    topics: Mapped[list[str]] = mapped_column(JSONB, default=list)
    # "metadata" is reserved on declarative classes, hence the attribute name.
    metadata_: Mapped[dict[str, Any]] = mapped_column("metadata", JSONB, default=dict)
    content_hash: Mapped[str] = mapped_column(String(64))
    # The stage run that fetched it. SET NULL: deleting a run must not delete evidence.
    produced_by: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("stage_runs.id", ondelete="SET NULL"), default=None
    )
```

```python
"""Integration fixtures. They need a migrated Postgres and never run implicitly.

Set TEST_DATABASE_URL to a database already at ``alembic upgrade head``. CI does
this against a throwaway service container. Each test runs inside a transaction
that is rolled back, so tests leave no rows behind.
"""

import os
from collections.abc import Iterator

import pytest
from sqlalchemy import create_engine
from sqlalchemy.orm import Session

from signalforge.db.session import normalise_dsn


@pytest.fixture
def db_session() -> Iterator[Session]:
    url = os.environ.get("TEST_DATABASE_URL")
    if not url:
        pytest.skip("TEST_DATABASE_URL not set")
    engine = create_engine(normalise_dsn(url))
    connection = engine.connect()
    outer = connection.begin()
    # create_savepoint: session.commit() inside the code under test releases a
    # savepoint, and the outer transaction is still rolled back at the end.
    session = Session(bind=connection, join_transaction_mode="create_savepoint")
    try:
        yield session
    finally:
        session.close()
        outer.rollback()
        connection.close()
        engine.dispose()
```

```python
import base64
import binascii
import uuid
from collections.abc import Sequence
from dataclasses import dataclass
from datetime import datetime

from sqlalchemy import select, tuple_
from sqlalchemy.dialects.postgresql import insert
from sqlalchemy.orm import Session

from signalforge.core.artifacts.signal import RawSignal
from signalforge.core.errors import ArtifactValidationError
from signalforge.db.models.signals import Signal


@dataclass(frozen=True)
class SignalPage:
    items: list[Signal]
    next_cursor: str | None


def _encode_cursor(signal: Signal) -> str:
    raw = f"{signal.published_at.isoformat()}|{signal.id}"
    return base64.urlsafe_b64encode(raw.encode()).decode()


def _decode_cursor(cursor: str) -> tuple[datetime, uuid.UUID]:
    try:
        published_at, signal_id = base64.urlsafe_b64decode(cursor.encode()).decode().split("|")
        return datetime.fromisoformat(published_at), uuid.UUID(signal_id)
    except (ValueError, binascii.Error, UnicodeDecodeError) as exc:
        raise ArtifactValidationError("Invalid cursor") from exc


class SignalRepository:
    def __init__(self, session: Session) -> None:
        self._session = session

    def insert_new(self, signals: Sequence[RawSignal]) -> int:
        """Insert signals with an unseen canonical URL and return how many were new.

        Signals are immutable, so a known URL is skipped, not updated. That makes
        re-running an ingest safe.
        """
        if not signals:
            return 0
        rows = [
            {
                "id": s.id,
                "schema_version": s.schema_version,
                "source": s.source,
                "source_type": s.source_type,
                "canonical_url": s.canonical_url,
                "title": s.title,
                "published_at": s.published_at,
                "summary": s.summary,
                "entities": list(s.entities),
                "topics": list(s.topics),
                "metadata": s.metadata,
                "content_hash": s.content_hash,
                "produced_by": s.produced_by,
            }
            for s in signals
        ]
        statement = (
            insert(Signal)
            .values(rows)
            .on_conflict_do_nothing(index_elements=["canonical_url"])
            .returning(Signal.id)
        )
        return len(self._session.execute(statement).all())

    def list_page(
        self, *, limit: int, cursor: str | None = None, source: str | None = None
    ) -> SignalPage:
        """Newest first. Keyset paging, so pages stay stable while new rows arrive."""
        query = select(Signal).order_by(Signal.published_at.desc(), Signal.id.desc())
        if source is not None:
            query = query.where(Signal.source == source)
        if cursor is not None:
            published_at, signal_id = _decode_cursor(cursor)
            query = query.where(
                tuple_(Signal.published_at, Signal.id) < tuple_(published_at, signal_id)
            )
        # One extra row tells us whether another page exists without a count query.
        rows = list(self._session.scalars(query.limit(limit + 1)))
        items = rows[:limit]
        next_cursor = _encode_cursor(items[-1]) if len(rows) > limit else None
        return SignalPage(items=items, next_cursor=next_cursor)
```

```python
"""Bounded retries with exponential backoff for external calls (standards §2)."""

import logging
import time
from collections.abc import Callable

from signalforge.core.errors import RateLimitedError, RetryableError

log = logging.getLogger(__name__)


def call_with_retry[T](
    fn: Callable[[], T],
    *,
    attempts: int = 3,
    base_delay: float = 0.5,
    sleep: Callable[[float], None] = time.sleep,
) -> T:
    """Call ``fn``, retrying only :class:`RetryableError` up to ``attempts`` times.

    A provider's ``Retry-After`` wins over the computed backoff. ``sleep`` is
    injectable so tests never wait.
    """
    if attempts < 1:
        raise ValueError("attempts must be at least 1")
    for attempt in range(1, attempts):
        try:
            return fn()
        except RetryableError as exc:
            delay = base_delay * 2 ** (attempt - 1)
            if isinstance(exc, RateLimitedError) and exc.retry_after_seconds is not None:
                delay = exc.retry_after_seconds
            log.warning(
                "retrying after error",
                extra={"attempt": attempt, "delay_s": delay, "error": str(exc)},
            )
            sleep(delay)
    return fn()
```

```python
"""Hacker News via the public Algolia search API. No key required.

The only module that knows this API's shape; the rest of the system sees
``RawSignal`` and the ``Source`` Protocol.
"""

import logging
import time
from collections.abc import Callable, Sequence
from datetime import UTC, datetime

import httpx
from pydantic import BaseModel, ConfigDict, ValidationError

from signalforge.core.artifacts.signal import RawSignal
from signalforge.core.errors import (
    ArtifactValidationError,
    OperationTimeoutError,
    ProviderRejectedError,
    ProviderUnavailableError,
    RateLimitedError,
)
from signalforge.core.retry import call_with_retry

log = logging.getLogger(__name__)

SEARCH_URL = "https://hn.algolia.com/api/v1/search_by_date"
REQUEST_TIMEOUT_S = 10.0
# ponytail: hard-coded until the vertical config (ai_engineering.yaml) exists.
DEFAULT_QUERIES: tuple[str, ...] = ("LLM", "AI agent", "RAG", "fine-tuning", "inference")


class _Hit(BaseModel):
    # Upstream schema, not ours: ignore keys we do not use instead of failing on them.
    model_config = ConfigDict(extra="ignore")

    objectID: str  # noqa: N815  (upstream field name)
    title: str
    url: str | None = None
    created_at_i: int
    points: int | None = None
    num_comments: int | None = None
    author: str | None = None


class _Response(BaseModel):
    model_config = ConfigDict(extra="ignore")

    hits: list[_Hit]


class HackerNewsSource:
    name = "hackernews"
    source_type = "community"

    def __init__(
        self,
        client: httpx.Client,
        *,
        queries: Sequence[str] = DEFAULT_QUERIES,
        min_points: int = 20,
        per_query: int = 50,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self._client = client
        self._queries = tuple(queries)
        self._min_points = min_points
        self._per_query = per_query
        self._sleep = sleep

    def fetch(self, since: datetime) -> list[RawSignal]:
        signals: dict[str, RawSignal] = {}
        for query in self._queries:
            for hit in self._search(query, since):
                signal = self._to_signal(hit)
                if signal is not None:
                    signals.setdefault(signal.canonical_url, signal)
        return list(signals.values())

    def _search(self, query: str, since: datetime) -> list[_Hit]:
        params: dict[str, str | int] = {
            "tags": "story",
            "query": query,
            "numericFilters": f"created_at_i>{int(since.timestamp())},points>={self._min_points}",
            "hitsPerPage": self._per_query,
        }
        return call_with_retry(lambda: self._get(params), sleep=self._sleep)

    def _get(self, params: dict[str, str | int]) -> list[_Hit]:
        try:
            response = self._client.get(SEARCH_URL, params=params, timeout=REQUEST_TIMEOUT_S)
        except httpx.TimeoutException as exc:
            raise OperationTimeoutError("Hacker News request timed out") from exc
        except httpx.TransportError as exc:
            raise ProviderUnavailableError("Hacker News is unreachable") from exc

        status = response.status_code
        if status == httpx.codes.TOO_MANY_REQUESTS:
            retry_after = response.headers.get("Retry-After", "")
            raise RateLimitedError(
                "Hacker News rate limited the request",
                retry_after_seconds=float(retry_after) if retry_after.isdigit() else None,
            )
        if status >= httpx.codes.INTERNAL_SERVER_ERROR:
            raise ProviderUnavailableError("Hacker News server error", context={"status": status})
        if status >= httpx.codes.BAD_REQUEST:
            raise ProviderRejectedError("Hacker News rejected the request", context={"status": status})
        try:
            return _Response.model_validate_json(response.content).hits
        except ValidationError as exc:
            raise ArtifactValidationError(
                "Unexpected Hacker News response shape", context={"errors": exc.error_count()}
            ) from exc

    def _to_signal(self, hit: _Hit) -> RawSignal | None:
        if not hit.url:  # Ask HN and similar posts have no external page to research.
            return None
        try:
            return RawSignal(
                source=self.name,
                source_type=self.source_type,
                canonical_url=hit.url,
                title=hit.title,
                published_at=datetime.fromtimestamp(hit.created_at_i, UTC),
                metadata={
                    "hn_id": hit.objectID,
                    "hn_url": f"https://news.ycombinator.com/item?id={hit.objectID}",
                    "points": hit.points,
                    "num_comments": hit.num_comments,
                    "author": hit.author,
                },
            )
        except (ArtifactValidationError, ValueError) as exc:
            # One malformed item must not fail the whole run, but it must not vanish silently.
            log.warning("skipping hit", extra={"hn_id": hit.objectID, "error": str(exc)})
            return None
```

```python
"""Ingest: fetch signals from a source and store the new ones.

A plain function so it can be called from a CLI, an HTTP trigger or, later, a
queue job without change. It records what it did in ``stage_runs`` (PRD §30).
"""

import hashlib
import logging
import uuid
from dataclasses import dataclass
from datetime import UTC, datetime

from sqlalchemy.orm import Session

from signalforge.core.logging import correlation_id
from signalforge.core.protocols import Source
from signalforge.db.models import StageRun, StageRunStatus
from signalforge.db.repositories.signals import SignalRepository

log = logging.getLogger(__name__)

INGEST_STAGE = "ingest"
_ENTITY_NAMESPACE = uuid.UUID("6f1c2f0e-3f0a-4b8e-9d55-0c6a6a1d7a01")
_MAX_ERROR_CHARS = 2000


@dataclass(frozen=True)
class IngestResult:
    source: str
    fetched: int
    inserted: int
    stage_run_id: uuid.UUID


def _sha(text: str) -> str:
    return hashlib.sha256(text.encode()).hexdigest()


def run_ingest(source: Source, session: Session, *, since: datetime) -> IngestResult:
    """Fetch from ``source`` since ``since``, store unseen signals, record the run.

    Failures leave a ``failed`` stage run with the reason and are re-raised.
    """
    started_at = datetime.now(UTC)
    run = StageRun(
        stage=INGEST_STAGE,
        entity_id=uuid.uuid5(_ENTITY_NAMESPACE, source.name),
        # ponytail: the start time makes every run unique, so the idempotency cache
        # never short-circuits a re-fetch. Real input/config hashing arrives with the
        # generic stage runner.
        input_hash=_sha(f"{source.name}|{since.isoformat()}|{started_at.isoformat()}"),
        stage_config_hash=_sha(source.name),
        status=StageRunStatus.RUNNING,
    )
    session.add(run)
    # Commit before doing any work: the run row must survive a rollback of the work.
    session.commit()
    run_id = run.id

    with correlation_id(str(run_id)):
        try:
            signals = list(source.fetch(since))
            stamped = [signal.derive(produced_by=run_id) for signal in signals]
            inserted = SignalRepository(session).insert_new(stamped)
            run.status = StageRunStatus.SUCCEEDED
            run.finished_at = datetime.now(UTC)
            run.output_schema_version = stamped[0].schema_version if stamped else None
            session.commit()
        except Exception as exc:
            session.rollback()
            failed = session.get(StageRun, run_id)
            if failed is not None:
                failed.status = StageRunStatus.FAILED
                failed.error = str(exc)[:_MAX_ERROR_CHARS]
                failed.finished_at = datetime.now(UTC)
                session.commit()
            log.exception("ingest failed", extra={"source": source.name})
            raise
        log.info(
            "ingest finished",
            extra={"source": source.name, "fetched": len(signals), "inserted": inserted},
        )
    return IngestResult(source=source.name, fetched=len(signals), inserted=inserted, stage_run_id=run_id)
```

```python
"""One consistent error shape for every failure (standards §2)."""

import logging
import re
from http import HTTPStatus

from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from starlette.exceptions import HTTPException as StarletteHTTPException

from signalforge.core.errors import ArtifactValidationError, RetryableError, SignalForgeError
from signalforge.core.logging import current_correlation_id

log = logging.getLogger(__name__)


def error_response(status: int, code: str, message: str) -> JSONResponse:
    body = {"error": {"code": code, "message": message, "correlation_id": current_correlation_id()}}
    return JSONResponse(body, status_code=status)


def _code_for(status: int) -> str:
    return re.sub(r"\W+", "_", HTTPStatus(status).phrase.lower())


def register_error_handlers(app: FastAPI) -> None:
    @app.exception_handler(RequestValidationError)
    async def _validation(_: Request, exc: RequestValidationError) -> JSONResponse:
        detail = "; ".join(f"{'.'.join(map(str, e['loc']))}: {e['msg']}" for e in exc.errors())
        return error_response(422, "validation_error", detail)

    @app.exception_handler(StarletteHTTPException)
    async def _http(_: Request, exc: StarletteHTTPException) -> JSONResponse:
        return error_response(exc.status_code, _code_for(exc.status_code), str(exc.detail))

    @app.exception_handler(ArtifactValidationError)
    async def _artifact(_: Request, exc: ArtifactValidationError) -> JSONResponse:
        return error_response(422, "validation_error", exc.message)

    @app.exception_handler(RetryableError)
    async def _retryable(_: Request, exc: RetryableError) -> JSONResponse:
        log.warning("temporary failure", extra={"error": str(exc)})
        return error_response(503, "temporarily_unavailable", "Temporarily unavailable, retry later")

    @app.exception_handler(SignalForgeError)
    async def _domain(_: Request, exc: SignalForgeError) -> JSONResponse:
        log.error("request failed", extra={"error": str(exc)})
        return error_response(500, "internal_error", "Internal error")

    @app.exception_handler(Exception)
    async def _unexpected(_: Request, __: Exception) -> JSONResponse:
        # Logged once, here, at the boundary. The client gets no internals.
        log.exception("unhandled error")
        return error_response(500, "internal_error", "Internal error")
```

```python
"""FastAPI app factory. Run: ``uv run uvicorn --factory apps.api.main:create_app``."""

import re
import uuid
from collections.abc import AsyncIterator, Awaitable, Callable
from contextlib import asynccontextmanager

from fastapi import FastAPI, Request, Response

from apps.api.errors import register_error_handlers
from apps.api.routes import signals
from signalforge.core.config import Settings, get_settings
from signalforge.core.logging import configure_logging, correlation_id
from signalforge.db.session import ConnectionMode, create_db_engine, create_session_factory

REQUEST_ID_HEADER = "X-Request-ID"
_SAFE_REQUEST_ID = re.compile(r"^[A-Za-z0-9._-]{1,64}$")


def create_app(settings: Settings | None = None) -> FastAPI:
    settings = settings or get_settings()
    configure_logging(settings.log_level)

    @asynccontextmanager
    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
        # Transaction-mode connection: the API's transactions are short (ADR-0003).
        engine = create_db_engine(settings, ConnectionMode.TRANSACTION)
        app.state.session_factory = create_session_factory(engine)
        try:
            yield
        finally:
            engine.dispose()

    app = FastAPI(title="SignalForge API", lifespan=lifespan)
    app.state.settings = settings
    register_error_handlers(app)

    @app.middleware("http")
    async def request_id(
        request: Request, call_next: Callable[[Request], Awaitable[Response]]
    ) -> Response:
        supplied = request.headers.get(REQUEST_ID_HEADER, "")
        rid = supplied if _SAFE_REQUEST_ID.match(supplied) else str(uuid.uuid4())
        with correlation_id(rid):
            response = await call_next(request)
        response.headers[REQUEST_ID_HEADER] = rid
        return response

    @app.get("/health")
    def health() -> dict[str, str]:
        return {"status": "ok"}

    app.include_router(signals.router)
    return app
```

Known trap: exceptions raised in an endpoint are handled by the exception handlers *inside* the middleware only for `HTTPException`-family and registered handlers; the catch-all `Exception` handler runs in Starlette's `ServerErrorMiddleware`, **outside** user middleware. That is why `test_invalid_cursor_maps_to_422` expects `correlation_id == X-Request-ID` (handled inside) but `test_unexpected_error…` only asserts the code. If the request id is missing from the 500 body, that is expected; do not work around it by swallowing exceptions in the middleware.

```python
"""Temporary HTTP trigger for ingest. Registered only when the flag is on."""

import secrets
from collections.abc import Callable
from datetime import UTC, datetime, timedelta
from typing import Annotated

import httpx
from fastapi import APIRouter, Depends, Header, HTTPException, Query, Request
from pydantic import BaseModel
from sqlalchemy.orm import Session

from apps.api.deps import get_session
from signalforge.plugins.sources import SOURCE_NAMES, USER_AGENT, build_source
from signalforge.stages.ingest import IngestResult, run_ingest

router = APIRouter()

IngestRunner = Callable[[str, int], IngestResult]


class IngestOut(BaseModel):
    source: str
    fetched: int
    inserted: int
    stage_run_id: str


def get_ingest_runner(session: Annotated[Session, Depends(get_session)]) -> IngestRunner:
    def run(source: str, hours: int) -> IngestResult:
        with httpx.Client(headers={"User-Agent": USER_AGENT}) as client:
            return run_ingest(
                build_source(source, client),
                session,
                since=datetime.now(UTC) - timedelta(hours=hours),
            )

    return run


def require_token(
    request: Request, x_ingest_token: Annotated[str | None, Header()] = None
) -> None:
    expected = request.app.state.settings.ingest_token
    supplied = x_ingest_token or ""
    # compare_digest: constant-time, so the token cannot be guessed byte by byte.
    if expected is None or not secrets.compare_digest(supplied, expected.get_secret_value()):
        raise HTTPException(status_code=401, detail="Invalid or missing ingest token")


@router.post("/ingest/{source}", response_model=IngestOut, dependencies=[Depends(require_token)])
def ingest(
    source: str,
    runner: Annotated[IngestRunner, Depends(get_ingest_runner)],
    hours: Annotated[int, Query(ge=1, le=24 * 14)] = 24,
) -> IngestOut:
    if source not in SOURCE_NAMES:
        raise HTTPException(status_code=404, detail=f"Unknown source: {source}")
    result = runner(source, hours)
    return IngestOut(
        source=result.source,
        fetched=result.fetched,
        inserted=result.inserted,
        stage_run_id=str(result.stage_run_id),
    )
```

```ts
export function formatRelative(now: Date, iso: string): string {
  const seconds = Math.max(0, (now.getTime() - new Date(iso).getTime()) / 1000);
  if (seconds < 60) return "just now";
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes}m ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  return `${Math.floor(hours / 24)}d ago`;
}

export function hostOf(url: string): string {
  try {
    return new URL(url).hostname.replace(/^www\./, "");
  } catch {
    return url;
  }
}

// Fetched content is untrusted: never render a link whose scheme is not http(s).
export function safeHref(url: string): string | null {
  try {
    const { protocol } = new URL(url);
    return protocol === "http:" || protocol === "https:" ? url : null;
  } catch {
    return null;
  }
}
```

```ts
export type Signal = {
  id: string;
  source: string;
  source_type: string;
  canonical_url: string;
  title: string;
  published_at: string;
  summary: string;
  metadata: Record<string, unknown>;
};

export type SignalList = { items: Signal[]; next_cursor: string | null };

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function parseSignal(value: unknown): Signal {
  if (!isRecord(value)) throw new Error("signal is not an object");
  const { id, source, source_type, canonical_url, title, published_at, summary, metadata } = value;
  if (
    typeof id !== "string" ||
    typeof source !== "string" ||
    typeof source_type !== "string" ||
    typeof canonical_url !== "string" ||
    typeof title !== "string" ||
    typeof published_at !== "string" ||
    typeof summary !== "string" ||
    !isRecord(metadata)
  ) {
    throw new Error("signal has an unexpected shape");
  }
  return { id, source, source_type, canonical_url, title, published_at, summary, metadata };
}

export function parseSignalList(json: unknown): SignalList {
  if (!isRecord(json) || !Array.isArray(json.items)) throw new Error("unexpected response shape");
  const next = json.next_cursor;
  if (next !== null && typeof next !== "string") throw new Error("unexpected cursor");
  return { items: json.items.map(parseSignal), next_cursor: next };
}

const PAGE_SIZE = 20;
const TIMEOUT_MS = 10_000;

export async function fetchSignals(cursor?: string): Promise<SignalList> {
  const base = process.env.API_BASE_URL;
  if (!base) throw new Error("API_BASE_URL is not set");
  const params = new URLSearchParams({ limit: String(PAGE_SIZE) });
  if (cursor) params.set("cursor", cursor);
  const response = await fetch(`${base}/signals?${params}`, {
    cache: "no-store",
    signal: AbortSignal.timeout(TIMEOUT_MS),
  });
  if (!response.ok) throw new Error(`API returned ${response.status}`);
  return parseSignalList(await response.json());
}
```
